\section{Methodology \& Architecture}
\begin{frame}{Swept Circular Arc Kinematics}
    \begin{block}{Discrete Pose Integration}
        Given displacement increments $\Delta s_L$ and $\Delta s_R$ over track gauge $W$:
        \begin{equation}
        \Delta \theta = \frac{\Delta s_R - \Delta s_L}{W}, \quad \Delta s = \frac{\Delta s_L + \Delta s_R}{2}
        \end{equation}
    \end{block}

    \begin{itemize}
        \item \textbf{Linear Displacement}: $x_{k+1} = x_k + \Delta s \cdot \cos(\theta_k + \Delta\theta / 2)$
        \item \textbf{Lateral Displacement}: $y_{k+1} = y_k + \Delta s \cdot \sin(\theta_k + \Delta\theta / 2)$
    \end{itemize}
\end{frame}
